1) SQL PARA VERIFICAR (correr en Supabase SQL Editor)
-----------------------------------------------------

-- Valores reales guardados por método de pago (¿qué hay en la BD?)
SELECT metodo_pago, estado_pago, estado, COUNT(*) AS total
FROM public.pedidos
GROUP BY metodo_pago, estado_pago, estado
ORDER BY metodo_pago, estado_pago, estado;

-- Últimos pedidos con su método y estado de pago
SELECT id_pedido, fecha_pedido, metodo_pago, estado, estado_pago, total
FROM public.pedidos
ORDER BY id_pedido DESC
LIMIT 20;

-- Trazabilidad de pagos (panel usa MANUAL-…, cliente usa CLIENTE-…, pasarela usa la referencia)
SELECT id_pago, id_pedido, metodo_pago, monto, referencia, estado, confirmado_por, fecha_pago
FROM public.pagos
ORDER BY id_pago DESC
LIMIT 20;

-- Notificaciones y puntos generados al pagar
SELECT id_notificacion, id_cliente, tipo, titulo, id_pedido, creada_en
FROM public.notificaciones
ORDER BY id_notificacion DESC
LIMIT 10;

-- Nota: si metodo_pago sale vacío, es porque el pedido se creó antes de la migración.
-- El panel agrupa esos como "sin dato" (solo aparecen con el filtro "Todos").
